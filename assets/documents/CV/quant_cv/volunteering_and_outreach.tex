\section{\sc Volunteering  And Outreach}



{\bf{Seattle Animal Shelter and Rabbit Ears Rescue}}
\begin{itemize}
\setlength\itemsep{0em}
\item[$\circ$] Helped feed, clean, and socialize animals to prepare
them for adoptions\\ as well as help traumatized animals return to normalcy
\item[$\circ$] Volunteered at adoption events to help find animals suitable future homes \\ and educate the local community on proper animal care
\end{itemize}

{\bf{Tent City Collective}}
\begin{itemize}
\setlength\itemsep{0em}
\item[$\circ$] Worked with local communities and legislators to advocate housing reform
\item[$\circ$] Volunteered for Tent City 3 and helped organize their stay at UW, the first\\ public university in the nation to agree to do so
\end{itemize}

{\bf{Camp Kesem}}
\begin{itemize}
\setlength\itemsep{0em}
\item[$\circ$] Helped run Camp Kesem, a summer camp for children whose parents are fighting cancer, helping them to process their emotions in a supportive environment.
\end{itemize}
